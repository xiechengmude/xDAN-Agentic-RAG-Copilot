# 搜索算子下游衔接方案

## 一、当前问题分析

### 1.1 生成的算子可能不被支持
```python
# 当前生成
"after:2025-04-02 before:2025-07-01"

# 问题：
1. BrightData可能不支持after/before算子
2. Google搜索的时间算子格式可能不同
3. 中文搜索引擎（百度）有自己的时间限定语法
```

### 1.2 下游系统差异

| 搜索系统 | 时间算子支持 | 正确格式 |
|---------|------------|----------|
| Google | after:, before: | after:2025-04-01 |
| 百度 | 不支持算子 | 使用时间关键词 |
| BrightData | 需要确认 | 可能需要参数传递 |

## 二、改进方案

### 2.1 多格式搜索算子生成

```python
class SearchOperatorGenerator:
    """搜索算子生成器 - 支持多种下游系统"""
    
    def generate_operators(self, time_range: Dict, target: str = 'google') -> Dict[str, Any]:
        """
        根据目标系统生成合适的搜索算子
        
        Args:
            time_range: 时间范围信息
            target: 目标系统 (google/baidu/brightdata/universal)
        """
        operators = {
            'google': self._generate_google_operators(time_range),
            'baidu': self._generate_baidu_operators(time_range),
            'brightdata': self._generate_brightdata_params(time_range),
            'universal': self._generate_universal_keywords(time_range)
        }
        
        return operators.get(target, operators['universal'])
    
    def _generate_google_operators(self, time_range: Dict) -> List[str]:
        """Google搜索算子"""
        operators = []
        
        # Google支持的时间算子
        start_date = time_range['start_date']
        end_date = time_range['end_date']
        
        # 使用daterange算子（更可靠）
        julian_start = self._to_julian_date(start_date)
        julian_end = self._to_julian_date(end_date)
        operators.append(f"daterange:{julian_start}-{julian_end}")
        
        # 备选：after/before算子
        operators.append(f"after:{start_date.strftime('%Y-%m-%d')}")
        operators.append(f"before:{end_date.strftime('%Y-%m-%d')}")
        
        return operators
    
    def _generate_baidu_operators(self, time_range: Dict) -> List[str]:
        """百度搜索 - 使用时间关键词而非算子"""
        keywords = []
        
        # 百度不支持时间算子，使用关键词
        year = time_range['start_date'].year
        month = time_range['start_date'].month
        
        # 添加年月关键词
        keywords.append(f"{year}年")
        keywords.append(f"{year}年{month}月")
        
        # 如果是季度，添加季度关键词
        if 'quarter' in time_range.get('type', ''):
            quarter = (month - 1) // 3 + 1
            keywords.append(f"{year}年第{quarter}季度")
            keywords.append(f"{year}Q{quarter}")
        
        return keywords
    
    def _generate_brightdata_params(self, time_range: Dict) -> Dict[str, Any]:
        """BrightData参数格式"""
        return {
            'date_from': time_range['start_date'].strftime('%Y-%m-%d'),
            'date_to': time_range['end_date'].strftime('%Y-%m-%d'),
            'time_filter': 'custom_range'
        }
    
    def _generate_universal_keywords(self, time_range: Dict) -> List[str]:
        """通用时间关键词 - 兼容所有系统"""
        keywords = []
        
        start_date = time_range['start_date']
        end_date = time_range['end_date']
        
        # 年份
        keywords.append(str(start_date.year))
        
        # 月份范围
        if start_date.month == end_date.month:
            keywords.append(f"{start_date.year}年{start_date.month}月")
        else:
            # 季度判断
            start_quarter = (start_date.month - 1) // 3 + 1
            end_quarter = (end_date.month - 1) // 3 + 1
            if start_quarter == end_quarter:
                keywords.append(f"{start_date.year}年第{start_quarter}季度")
                keywords.append(f"{start_date.year}Q{start_quarter}")
        
        return keywords
```

### 2.2 下游系统适配层

```python
class SearchSystemAdapter:
    """搜索系统适配器"""
    
    def adapt_query_for_system(self, query: str, time_operators: Dict, 
                              system: str = 'brightdata') -> Dict[str, Any]:
        """
        为特定系统适配查询
        
        Args:
            query: 原始查询
            time_operators: 时间算子信息
            system: 目标系统
        """
        if system == 'brightdata':
            return self._adapt_for_brightdata(query, time_operators)
        elif system == 'google_direct':
            return self._adapt_for_google(query, time_operators)
        elif system == 'baidu':
            return self._adapt_for_baidu(query, time_operators)
        else:
            return {'query': query}
    
    def _adapt_for_brightdata(self, query: str, time_operators: Dict) -> Dict[str, Any]:
        """BrightData适配"""
        # BrightData可能支持参数化时间过滤
        adapted = {
            'query': query,  # 不在查询中添加时间算子
            'filters': {
                'time_range': time_operators.get('brightdata', {})
            }
        }
        
        # 添加时间关键词增强相关性
        universal_keywords = time_operators.get('universal', [])
        if universal_keywords:
            adapted['query'] = f"{query} {' '.join(universal_keywords[:2])}"
        
        return adapted
    
    def _adapt_for_google(self, query: str, time_operators: Dict) -> Dict[str, Any]:
        """Google搜索适配"""
        google_ops = time_operators.get('google', [])
        
        # 使用最可靠的算子
        if google_ops:
            # 优先使用daterange
            daterange_op = next((op for op in google_ops if 'daterange:' in op), None)
            if daterange_op:
                return {'query': f"{daterange_op} {query}"}
            else:
                # 使用after/before
                return {'query': f"{google_ops[0]} {google_ops[1]} {query}"}
        
        return {'query': query}
```

### 2.3 集成到搜索策略

```python
# 在 search_strategy.py 中集成
class EnhancedSearchStrategy(SearchStrategy):
    """增强的搜索策略 - 考虑下游系统"""
    
    def __init__(self):
        super().__init__()
        self.operator_generator = SearchOperatorGenerator()
        self.system_adapter = SearchSystemAdapter()
    
    def optimize_for_system(self, question: str, target_system: str = 'brightdata') -> Dict[str, Any]:
        """
        为特定系统优化查询
        
        Args:
            question: 原始问题
            target_system: 目标搜索系统
        """
        # 1. 基础分析
        analysis = self.analyze_question(question)
        
        # 2. 时间解析
        time_refs = self.time_aware.parse_relative_time(question)
        
        # 3. 生成系统特定的算子
        time_operators = {}
        if time_refs:
            time_range = time_refs[0]['range']
            time_operators = self.operator_generator.generate_operators(
                time_range, 
                target=target_system
            )
        
        # 4. 适配查询
        adapted_query = self.system_adapter.adapt_query_for_system(
            question, 
            time_operators, 
            target_system
        )
        
        return {
            'original': question,
            'adapted': adapted_query,
            'system': target_system,
            'time_operators': time_operators,
            'analysis': analysis
        }
```

async def search(self, question: str, use_alternative_strategy: bool = False):

## 五、最佳实践

### 5.1 算子选择策略

1. **优先使用通用关键词**：确保基本兼容性
2. **系统特定优化**：针对已知系统使用最优算子
3. **降级处理**：算子不支持时回退到关键词

### 5.2 时间表达优先级

```
1. 时间关键词（2025年6月）- 最通用
2. 时间范围参数（date_from/date_to）- API支持时最佳
3. 搜索算子（after:/before:）- Google等支持
4. 高级算子（daterange:）- 特定系统优化
```

## 六、总结

通过考虑下游系统的衔接：

1. ✅ 为不同系统生成合适的算子格式
2. ✅ 提供降级方案确保兼容性
3. ✅ 支持参数化时间过滤（API最友好）
4. ✅ 保留通用时间关键词作为后备
5. ✅ 系统适配层处理差异
