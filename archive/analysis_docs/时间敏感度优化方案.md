# 时间敏感度优化方案 - "最近"语义理解问题

## 一、当前问题分析

### 1.1 现有实现的缺陷
```python
# 当前实现 - 过于简单
if '最近' in question:
    enhanced_question = f"{question} 2025"  # 仅添加当前年份
```

**问题**：
- "最近"被处理成了"2025年"，这可能包含了未来6个月的数据
- 没有考虑"最近"的实际时间范围（通常指3-6个月）
- 2025年7月1日，搜索"最近"应该是2025年1月-6月的数据

### 1.2 正确的时间理解

当前时间：2025年7月1日

| 时间词 | 应该对应的范围 | 当前实现 | 正确实现 |
|--------|--------------|----------|----------|
| 最近 | 2025年4月-6月（Q2） | 2025年全年 | 2025 Q2 |
| 最新 | 2025年6月/Q2最新数据 | 2025年 | 2025年6月 |
| 近期 | 2025年1月-6月 | 2025年 | 2025 H1 |
| 今年 | 2025年1月-6月 | 2025年 | 2025 YTD |

### 2.1 增强时间理解逻辑

```python
from datetime import datetime, timedelta

def enhance_time_context_smart(question: str, analysis: Dict[str, Any]) -> str:
    """
    智能时间上下文增强
    """
    current_date = datetime.now()  # 2025-07-01
    current_year = current_date.year
    current_month = current_date.month
    current_quarter = (current_month - 1) // 3 + 1  # Q3
    
    # 相对时间词映射
    relative_time_map = {
        '最近': {
            'range': 3,  # 最近3个月
            'unit': 'month',
            'query': f'("{current_year}年{current_month-2}月" OR "{current_year}年{current_month-1}月" OR "{current_year}年{current_month}月")'
        },
        '最新': {
            'range': 1,  # 最新1个月
            'unit': 'month', 
            'query': f'"{current_year}年{current_month}月" latest'
        },
        '近期': {
            'range': 6,  # 近期6个月
            'unit': 'month',
            'query': f'"{current_year}上半年" OR "{current_year} H1"'
        },
        '今年': {
            'range': current_month,  # 今年至今
            'unit': 'month',
            'query': f'"{current_year}年" YTD'
        }
    }
    
    # 检测并处理相对时间
    for time_word, config in relative_time_map.items():
        if time_word in question:
            # 财报特殊处理 - 使用季度
            if '财报' in question or '财务' in question:
                if time_word == '最近':
                    # 最近的完整季度
                    last_quarter = current_quarter - 1 if current_quarter > 1 else 4
                    last_quarter_year = current_year if current_quarter > 1 else current_year - 1
                    query_enhancement = f'"{last_quarter_year}年第{last_quarter}季度" OR "{last_quarter_year} Q{last_quarter}"'
                elif time_word == '最新':
                    # 最新可能是当前季度或上一季度
                    query_enhancement = f'"{current_year}年第{current_quarter}季度" OR "{current_year} Q{current_quarter}"'
            else:
                query_enhancement = config['query']
            
            return f"{question} {query_enhancement}"
    
    # 默认处理
    return f"{question} {current_year}"
```

### 2.2 搜索算子优化

```python
def add_time_operators(question: str, analysis: Dict[str, Any]) -> List[str]:
    """
    添加时间相关的搜索算子
    """
    operators = []
    current_date = datetime.now()
    
    if '最近' in question:
        # 最近3个月
        three_months_ago = current_date - timedelta(days=90)
        operators.append(f'after:{three_months_ago.strftime("%Y-%m-%d")}')
        operators.append(f'before:{current_date.strftime("%Y-%m-%d")}')
        
    elif '最新' in question:
        # 最近1个月
        one_month_ago = current_date - timedelta(days=30)
        operators.append(f'after:{one_month_ago.strftime("%Y-%m-%d")}')
        
    elif '今年' in question:
        # 今年至今
        operators.append(f'after:{current_date.year}-01-01')
        operators.append(f'before:{current_date.strftime("%Y-%m-%d")}')
        
    return operators
```

class SearchStrategy:
    def __init__(self):
        # 添加相对时间配置
        self.relative_time_config = {
            '最近': {'months': 3, 'prefer_quarter': True},
            '最新': {'months': 1, 'prefer_latest': True},
            '近期': {'months': 6, 'prefer_half_year': True},
            '今年': {'months': 'ytd', 'prefer_ytd': True}
        }
        
    def parse_relative_time(self, question: str) -> Dict[str, Any]:
        """解析相对时间词"""
        current_date = datetime.now()
        
        for time_word, config in self.relative_time_config.items():
            if time_word in question:
                if config.get('prefer_quarter') and ('财报' in question or '财务' in question):
                    # 返回季度范围
                    current_quarter = (current_date.month - 1) // 3 + 1
                    if current_quarter == 1:  # Q1，返回去年Q4
                        return {
                            'year': current_date.year - 1,
                            'quarter': 4,
                            'time_range': f'{current_date.year - 1} Q4'
                        }
                    else:  # 返回上一季度
                        return {
                            'year': current_date.year,
                            'quarter': current_quarter - 1,
                            'time_range': f'{current_date.year} Q{current_quarter - 1}'
                        }
                        
                # 其他时间处理...
        
        return None

1. ✅ 准确理解"最近"的时间范围
